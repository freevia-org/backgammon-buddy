cmake_minimum_required(VERSION 3.22)
project(buddy_gnubg VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(EXPAT_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(EXPAT_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(EXPAT_DTD OFF CACHE BOOL "" FORCE)
set(EXPAT_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/expat-2.7.5 EXCLUDE_FROM_ALL)
option(BUDDY_GNUBG_STATIC "Build for static iOS linkage" OFF)
set(BUDDY_LIBRARY_TYPE SHARED)
if(BUDDY_GNUBG_STATIC)
 set(BUDDY_LIBRARY_TYPE STATIC)
endif()
add_library(buddy_gnubg ${BUDDY_LIBRARY_TYPE}
 upstream/eval.c upstream/bearoff.c upstream/bearoffgammon.c
 upstream/positionid.c upstream/matchequity.c upstream/mec.c
 upstream/lib/neuralnet.c upstream/lib/cache.c upstream/lib/isaac.c
 upstream/lib/md5.c upstream/lib/inputs.c upstream/lib/list.c
 portable/glib_compat.c buddy_gnubg.cpp)
target_include_directories(buddy_gnubg PRIVATE portable upstream upstream/lib)
target_compile_definitions(buddy_gnubg PRIVATE _GNU_SOURCE)
target_link_libraries(buddy_gnubg PRIVATE expat m)
set_target_properties(buddy_gnubg PROPERTIES CXX_STANDARD 17)
target_compile_options(buddy_gnubg PRIVATE -ffunction-sections -fdata-sections)
set_target_properties(buddy_gnubg PROPERTIES C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden)
if(APPLE)
 target_link_options(buddy_gnubg PRIVATE "-Wl,-dead_strip")
else()
 target_link_options(buddy_gnubg PRIVATE "-Wl,--gc-sections")
endif()
if(WIN32)
 target_link_options(buddy_gnubg PRIVATE "-Wl,--exclude-all-symbols" -static -static-libgcc -static-libstdc++)
endif()
if(ANDROID)
 target_link_options(buddy_gnubg PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--no-undefined")
endif()
